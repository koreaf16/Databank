# 새 메뉴 추가 가이드

9개 표준 메뉴 외에 새 메뉴를 추가하거나, 기존 메뉴 안에 새 도메인을 끼워넣을 때의 표준 절차.

## 7-task 시퀀스 (Phase 1 패턴)

| Task | 내용 |
|---|---|
| x.1 DDL | `backend/src/modules/<menu>/schema/<menu>Schema.js` — `ensureXxxSchema(connection)` 함수 |
| x.2 Repository | `backend/src/modules/<menu>/repository/*Repository.js` — `repoFn(connection, payload)` |
| x.3 Routes | `backend/src/modules/<menu>/routes/*Routes.js` — Express Router |
| x.4 API 클라이언트 | `frontend/src/features/<menu>/api/<menu>Api.js` |
| x.5 UI 연결 | `frontend/src/features/<menu>/components/*` — apiClient 사용, window 글로벌 금지 |
| x.6 기술서 | `docs/modules/<menu>/README.md` — 8섹션 표준 |
| x.7 E2E 검증 | curl + UI 클릭 시퀀스 |

## 1. DDL 작성

```javascript
/**
 * 파일: backend/src/modules/example/schema/exampleSchema.js
 * 역할: example 도메인 테이블 DDL.
 *
 * 연관 파일:
 *   - ../repository/exampleRepository.js : 본 스키마 SELECT/INSERT
 *   - infra/oracle/errorCodes.js         : ORA-00955 swallow
 */
const oracledb = require('oracledb');
const { isAlreadyExists } = require('../../../infra/oracle/errorCodes');

const TABLES = [
  `CREATE TABLE EXAMPLES (
     EXAMPLE_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY,
     NAME VARCHAR2(200 CHAR) NOT NULL,
     ENABLED NUMBER(1) DEFAULT 1 NOT NULL,
     CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
     UPDATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
     CONSTRAINT PK_EXAMPLES PRIMARY KEY (EXAMPLE_ID)
   )`,
];

const MIGRATION_DDL = [
  // 신규 컬럼 ALTER 추가 시 여기에
];

async function ensureExampleSchema(connection) {
  for (const ddl of TABLES) {
    try { await connection.execute(ddl); }
    catch (err) { if (!isAlreadyExists(err)) throw err; }
  }
  for (const ddl of MIGRATION_DDL) {
    try { await connection.execute(ddl); }
    catch (err) { /* swallow */ }
  }
  await connection.commit();
}

module.exports = { ensureExampleSchema };
```

## 2. Repository

```javascript
/**
 * 파일: backend/src/modules/example/repository/exampleRepository.js
 * 역할: EXAMPLES 테이블 CRUD.
 */
const oracledb = require('oracledb');
const { clobFetchHandler, requiredText } = require('../../../infra/oracle/helpers');
const { validationError, notFoundError } = require('../../../http/errors');

async function listExamples(connection) {
  const result = await connection.execute(
    `SELECT EXAMPLE_ID, NAME, ENABLED FROM EXAMPLES WHERE ENABLED = 1 ORDER BY EXAMPLE_ID`,
    {},
    { outFormat: oracledb.OUT_FORMAT_OBJECT }
  );
  return result.rows.map(normalize);
}

async function createExample(connection, payload) {
  const name = requiredText(payload.name, 'name');
  const result = await connection.execute(
    `INSERT INTO EXAMPLES (NAME) VALUES (:name) RETURNING EXAMPLE_ID INTO :id`,
    { name, id: { type: oracledb.NUMBER, dir: oracledb.BIND_OUT } }
  );
  return { exampleId: result.outBinds.id[0], name };
}

function normalize(row) {
  return { exampleId: row.EXAMPLE_ID, name: row.NAME, enabled: row.ENABLED === 1 };
}

module.exports = { listExamples, createExample };
```

## 3. Routes

```javascript
/**
 * 파일: backend/src/modules/example/routes/exampleRoutes.js
 * 역할: /api/examples HTTP 라우트.
 */
const express = require('express');
const { withConnection } = require('../../../infra/oracle/withConnection');
const { withTransaction } = require('../../../infra/oracle/withTransaction');
const { asyncHandler } = require('../../../http/asyncHandler');
const { ok, created } = require('../../../http/apiResponse');
const repo = require('../repository/exampleRepository');

const router = express.Router();

router.get('/', asyncHandler(async (req, res) => {
  const data = await withConnection(conn => repo.listExamples(conn));
  res.json(ok(data));
}));

router.post('/', asyncHandler(async (req, res) => {
  const data = await withTransaction(conn => repo.createExample(conn, req.body));
  res.status(201).json(created(data));
}));

module.exports = router;
```

`backend/src/app.js`에 마운트:
```javascript
app.use('/api/examples', require('./modules/example/routes/exampleRoutes'));
```

```bash

  -H "Content-Type: application/json" \
  -d '{"name":"검증 항목"}'

curl -s http://127.0.0.1:7001/api/examples | jq '.data[0]'
```
